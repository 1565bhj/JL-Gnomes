2、二进制数据包协议
| 4字节头部  + 1字节请求类型  + 1字节数据类型  + 2字节总分片数 + 2字节分片序号 + 4字数据和校验 + 4字数据节长度 +   n字数据       | 
|   4BYTE    |   1BYTE        |   1BYTE        |   2BYTE       |   2BYTE       |   4BYTE       |   4BYTE       |     nBYTE       | 
|  "qyai"    |  req_type      |  data_type     |   total_seq   |   seq         |   data_sum    |    len        |data[0]...data[n]|

//服务器端下发+设备端上传[DATA过程]
struct ai_multimodal_request
{
	unsigned char head[4]; 		//"qyai" 
	unsigned char data_type;		//0 PCM音频，1 text纯文本，2 json，3 ogg-opus,  4 jpg图片，5 gif视频，6 avi视频 ，7...预留
	unsigned short total_seq;		//总的分片数据数量
	unsigned short seq;			//当前分片数据包序号，0 1 2 3.....
	unsigned int data_sum; 		//数据和，使用去校验数据完整性，如果data_sum = 0，则不使用和校验
	unsigned int len;      			//数据长度
	unsigned char data[n]; 		//len个字节数据
};

//设备端+服务器端回复[ACK]
struct ai_multimodal_ack
{
	unsigned char head[4]; 		//"ack" 
	unsigned char data_type;		//0 PCM音频，  1 text纯文本，2 json，3 ogg,  4 jpg图片，5 gif视频，6 avi视频 ，7...预留
	unsigned char respond_type;		//0 正常接收， 1 校验出错， 2 重新对话，3 请求关闭， 4...预留
	unsigned short seq;			//当前分片数据包序号，0 1 2 3.....
};
