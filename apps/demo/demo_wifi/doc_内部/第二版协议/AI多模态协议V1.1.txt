//请求类型
Multimodal_Type: audio/text-img/img-img/text&img-img/text-video/img-video/video-video, 320x240

//音视频支持类型，音频mp3或者ogg_opus， 图片jpg，视频gif(avi)

Dialogue_Type: vad/key

Explicit_Language：zh/en/ja/es-ms/id/pt-br


2、二进制数据包协议
4字节头部  + 1字节操作码类型 + 1字节数据类型  + 2字节分片序号 + 4字时间戳 + 4字数据校验 + 4字数据节长度 + n字数据   
该协议：数据内可以嵌套数据协议。


#define SEND_HEAD 0xFFFAE8D9

4字节对齐
----------------------------------------   
| 0xFF   |   0XFA    |  0XE8  |  0XD9  |   //头部：HEAD
----------------------------------------
| opcode | data_type |       seq       |   //1字节操作码类型 + 1字节数据类型 + 2字节分片序列号
----------------------------------------
|                ssrc                  |   //4字节随机ssrc, 用于确定回复数据源
----------------------------------------
|             time_stamp               |   //4字节时间戳
----------------------------------------
|              data_check              |   //4字节数据校验
----------------------------------------
|              len                     |   //4字节数据长度
----------------------------------------

//服务器下发：24字节
struct ai_multimodal_head
{
	unsigned char head[4]; 			//0xFFFAE8D9
	unsigned char opcode;			//操作码BIT(0)-BIT(7)， BIT(0)：fin-是否最后一个包（0否，1是）， BIT(1)：校验版类型（0 SUM， 1 CRC32），BIT(2)：数据内容是否含嵌套协议（0无，1有），BIT(2)-BIT(7)：预留
	unsigned char data_type;		//0 PCM音频，1 text纯文本，2 json，3 ogg-opus,  4 jpg图片，5 gif视频，6 avi视频 ，7...预留
	unsigned short seq;				//当前分片数据包序号，0 1 2 3.....
	unsigned int ssrc;				//4字节随机ssrc
	unsigned int time_stamp;		//预留：时间戳
	unsigned int data_check; 		//数据校验，使用去校验数据完整性，data_check = 0，则不使用校验：校验包括：数据+长度
	unsigned int len;      			//数据长度
	unsigned char data[n]; 			//len个字节数据
};



#define RESP_HEAD 0xFFFAE8DA

4字节对齐
-------------------------------------------- 
| 0xFF      |   0XFA       | 0XE8  | 0XDA  |   //头部：HEAD
--------------------------------------------
| data_type | respond_type |      seq      |   //1字节数据类型 + 1字节回复类型 + 2字节分片序列号
--------------------------------------------
|                 ssrc                     |   //4字节请求的ssrc值
--------------------------------------------

//设备回复：12字节
struct ai_multimodal_head_respond
{
	unsigned char head[4]; 			//0xFFFAE8DA 
	unsigned char data_type;		//0 PCM音频，  1 text纯文本，2 json，3 ogg,  4 jpg图片，5 gif视频，6 avi视频 ，7-15...预留
	unsigned char respond_type;		//0 正常接收， 1 校验出错， 2 重新对话，3 请求关闭， 4-15...预留
	unsigned short seq;				//当前分片数据包序号，0 1 2 3.....
	unsigned int ssrc;				//4字节请求的ssrc值
};
